Java Code

This is my Java program for Problem Number 167, Second Largest Number Finder. It uses a one-dimensional array to store 15 integers, finds the largest and second-largest distinct values, and arranges the numbers in descending order. 

import java.util.Scanner;

public class SecondLargestNumberFinder {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        int[] numbers = new int[15];

        System.out.println("SECOND LARGEST NUMBER FINDER");
        System.out.println("Enter 15 numbers from 0 to 100.");

        for (int i = 0; i < numbers.length; i++) {
            while (true) {
                System.out.print("Enter number " + (i + 1) + ": ");

                if (input.hasNextInt()) {
                    int value = input.nextInt();

                    if (value >= 0 && value <= 100) {
                        numbers[i] = value;
                        break;
                    } else {
                        System.out.println("Invalid number. Please enter a value from 0 to 100.");
                    }
                } else {
                    System.out.println("Invalid input. Please enter a whole number.");
                    input.next();
                }
            }
        }

        int largest = -1;
        int secondLargest = -1;

        for (int i = 0; i < numbers.length; i++) {
            if (numbers[i] > largest) {
                secondLargest = largest;
                largest = numbers[i];
            } else if (numbers[i] < largest &&
                       numbers[i] > secondLargest) {
                secondLargest = numbers[i];
            }
        }

        for (int i = 0; i < numbers.length - 1; i++) {
            for (int j = 0; j < numbers.length - 1 - i; j++) {
                if (numbers[j] < numbers[j + 1]) {
                    int temp = numbers[j];
                    numbers[j] = numbers[j + 1];
                    numbers[j + 1] = temp;
                }
            }
        }

        System.out.println("\\n===== RESULTS =====");
        System.out.println("Largest number: " + largest);

        if (secondLargest == -1) {
            System.out.println("No second-largest distinct value exists.");
        } else {
            System.out.println("Second-largest number: " + secondLargest);
        }

        System.out.println("\\nNumbers in descending order:");

        for (int i = 0; i < numbers.length; i++) {
            System.out.print(numbers[i] + " ");
        }

        System.out.println();
        input.close();
    }
}
